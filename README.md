<script>

async function copyTemplate(button) {

    /* 
       Uzimamo SAMO .copy-content.
       Naslov šablona i dugme se NE kopiraju.
    */

    const template = button.parentElement.querySelector(".copy-content");

    const html = template.innerHTML;

    /*
       Pravljenje plain-text verzije kao rezervne
       varijante za programe koji ne primaju HTML.
    */

    const temporary = document.createElement("div");
    temporary.innerHTML = html;

    const plainText = temporary.innerText;

    try {

        /*
           Kopiramo i HTML i običan tekst.
           Word će koristiti HTML verziju.
        */

        await navigator.clipboard.write([
            new ClipboardItem({
                "text/html": new Blob(
                    [html],
                    { type: "text/html" }
                ),

                "text/plain": new Blob(
                    [plainText],
                    { type: "text/plain" }
                )
            })
        ]);

        button.textContent = "✓ COPIED";
        button.classList.add("copied");

        setTimeout(() => {
            button.textContent = "📋 COPY TO CLIPBOARD";
            button.classList.remove("copied");
        }, 1500);

    } catch (error) {

        /*
           Rezervna opcija.
        */

        const range = document.createRange();

        range.selectNodeContents(template);

        const selection = window.getSelection();

        selection.removeAllRanges();
        selection.addRange(range);

        document.execCommand("copy");

        selection.removeAllRanges();

        button.textContent = "✓ COPIED";
        button.classList.add("copied");

        setTimeout(() => {
            button.textContent = "📋 COPY TO CLIPBOARD";
            button.classList.remove("copied");
        }, 1500);
    }
}

</script>
